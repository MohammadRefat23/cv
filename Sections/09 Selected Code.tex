\section{Selected Code}

\begin{compact}
\noindent\textbf{jhelum-stellar-stream} \githubrepo{https://github.com/MohammadRefat23/jhelum-stellar-stream}\hfill 2026\par
\noindent{\small Documented, tested Python package that rebuilds the Jhelum chemical-abundance analysis from public SDSS DR17 and VizieR data.}\par
\vspace{0.3em}
\noindent\textbf{StarryStarSpots\_MCMC} \githubrepo{https://github.com/MohammadRefat23/StarryStarSpots_MCMC}\hfill 2023\par
\noindent{\small Course project (Computation Methods in Physics): MCMC fitting of starspot parameters with starry and emcee.}\par
\vspace{0.3em}
\noindent\textbf{Spherical Harmonics Explorer} \researchlink{Tool}{https://mohammadrefat23.github.io/projects/spherical-harmonics/}\hfill 2026\par
\noindent{\small Interactive browser tool with a custom HTML Canvas renderer for exploring spherical-harmonic surface patterns, the basis of the brown-dwarf surface maps.}\par
\end{compact}
