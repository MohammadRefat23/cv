% Short job entry.
% #1 = title
% #2 = dates
\newenvironment{jobshort}[2]
{%
  \begin{tabularx}{\linewidth}{@{}l X r@{}}
    \textbf{#1} & \hfill & #2 \\[3.75pt]
  \end{tabularx}
}
{}

% Long job entry with an itemized description.
% #1 = title
% #2 = dates
\newenvironment{joblong}[2]
{%
  \begin{tabularx}{\linewidth}{@{}l X r@{}}
    \textbf{#1} & \hfill & #2 \\[3.75pt]
  \end{tabularx}

  \begin{minipage}[t]{\linewidth}
  \begin{itemize}[
    leftmargin=1em,
    label={--},
    itemsep=3pt,
    topsep=0pt,
    parsep=0pt,
    partopsep=0pt
  ]
}
{%
  \end{itemize}
  \end{minipage}
}

% Research experience heading
% #1 = project title
% #2 = institution
% #3 = advisor
% #4 = dates
% #5 = research output icons/links
\newcommand{\researchentry}[5]{%
  \Needspace{8\baselineskip}%
  \par\noindent

  % Project title followed by optional research-output icons
  \textbf{#1}%
  \if\relax\detokenize{#5}\relax
  \else
    \hspace{0.3em}{\normalfont #5}%
  \fi
  \par\nobreak

  \vspace{-0.15em}

  \noindent
  \begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}X
                                  >{\raggedleft\arraybackslash}l@{}}
    \textit{#2} & \mbox{\textit{#4}} \\[-0.15em]
    \textit{Advisor: #3} &
  \end{tabularx}

  \par\vspace{-0.4em}
}

% Presentation entry.
% #1 = title
% #2 = venue
% #3 = location/type
% #4 = date
\newenvironment{presentation}[4]
{%
  \Needspace{4\baselineskip}%
  \par\noindent
  \begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}Xr@{}}
    \textbf{#1} & \mbox{#4} \\[-0.1em]
    \multicolumn{2}{@{}X@{}}{%
      {\small\textit{#2} \enspace---\enspace #3}%
    }
  \end{tabularx}
  \par\vspace{0.1em}
}
{}

% Teaching appointment entry.
% #1 = position
% #2 = institution
% #3 = dates
\newenvironment{teaching}[3]
{%
  \begin{tabularx}{\linewidth}{@{}Xr@{}}
    \textbf{#1} & #3 \\
    \textit{#2} &
  \end{tabularx}

  \vspace{-0.5em}

  \begin{itemize}[
    leftmargin=1em,
    label={--},
    topsep=0pt,
    partopsep=0pt,
    parsep=0pt,
    itemsep=1.5pt
  ]
}
{%
  \end{itemize}
  \vspace{0.8em}
}

% Individual course entry used inside the teaching environment.
% #1 = course title/number
% #2 = term(s) taught
\newcommand{\teachingcourse}[2]{%
  \item
  \begin{tabularx}{\linewidth}{@{}Xr@{}}
    #1 & #2
  \end{tabularx}%
}

% Professional experience entry.
% #1 = position
% #2 = institution
% #3 = dates
\newenvironment{workexperience}[3]
{%
  \par\noindent
  \begin{tabularx}{\linewidth}{@{}Xr@{}}
    \textbf{#1} & \emph{#3} \\
    \textit{#2} &
  \end{tabularx}

  \vspace{-0.45em}

  \begin{itemize}[
    leftmargin=1.2em,
    label={--},
    itemsep=2pt,
    topsep=2pt,
    parsep=0pt,
    partopsep=0pt
  ]
}
{%
  \end{itemize}
  \vspace{0.4em}
}
